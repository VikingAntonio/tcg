-- Actualización para Soporte de Registro y Tiendas en Landing Page
-- Ejecuta esto en el Editor SQL de Supabase

-- 1. Agregar columna para identificar si el usuario es una tienda
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS is_store BOOLEAN DEFAULT false;

-- 2. Agregar columna para el correo electrónico
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS email TEXT;

-- Agregar columnas para el Perfil de Negocio
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS horario TEXT;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS ubicacion TEXT;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS store_logo TEXT;

-- 3. Asegurar que existan las columnas de límites (si no se agregaron antes)
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS max_albums INTEGER DEFAULT 3;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS max_pages INTEGER DEFAULT 5;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS max_decks INTEGER DEFAULT 1;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS max_cards_per_deck INTEGER DEFAULT 60;
ALTER TABLE usuarios ADD COLUMN IF NOT EXISTS allowed_spirit_ids TEXT DEFAULT '';

-- 4. Opcional: Actualizar usuarios existentes para que no tengan valores nulos
UPDATE usuarios SET is_store = false WHERE is_store IS NULL;
UPDATE usuarios SET max_albums = 3, max_pages = 5, max_decks = 1, max_cards_per_deck = 60 WHERE max_albums IS NULL;

-- 5. Activar tiendas conocidas (Basado en imagen proporcionada)
UPDATE usuarios SET is_store = true WHERE username IN ('Mexplay', 'toonShop');

-- 6. Configuración de Storage para Logos (Ejecutar en Editor SQL)
-- Esto crea el bucket 'logos' y lo hace público para que se puedan ver las imágenes
INSERT INTO storage.buckets (id, name, public)
VALUES ('logos', 'logos', true)
ON CONFLICT (id) DO NOTHING;

-- Políticas de Storage (Permitir a usuarios autenticados subir logos)
CREATE POLICY "Permitir subida de logos a usuarios autenticados"
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (bucket_id = 'logos');

CREATE POLICY "Permitir ver logos a todos"
ON storage.objects FOR SELECT
TO public
USING (bucket_id = 'logos');
