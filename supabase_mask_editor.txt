
-- SQL to add necessary columns for the Mask Editor and External Search
-- Run this in your Supabase SQL Editor

-- Add custom_mask_url to card_slots
ALTER TABLE card_slots ADD COLUMN IF NOT EXISTS custom_mask_url TEXT;

-- Add custom_mask_url to deck_cards
ALTER TABLE deck_cards ADD COLUMN IF NOT EXISTS custom_mask_url TEXT;

-- Add colors to albums (if not already present)
ALTER TABLE albums ADD COLUMN IF NOT EXISTS cover_color TEXT DEFAULT '#1a1a1a';
ALTER TABLE albums ADD COLUMN IF NOT EXISTS back_color TEXT DEFAULT '#1a1a1a';
