-- SQL para gestión restringida de usuarios
-- Solo los administradores pueden modificar campos sensibles

-- 1. Política para permitir que los admins actualicen cualquier perfil
-- Nota: La política existente "Users can update their own profile" se mantiene
-- para permitir que los usuarios normales sigan editando su propio perfil (limitado por el trigger).

DROP POLICY IF EXISTS "Admins can update any profile" ON public.usuarios;
CREATE POLICY "Admins can update any profile" ON public.usuarios
    FOR UPDATE
    TO authenticated
    USING (
        (SELECT role FROM public.usuarios WHERE id = auth.uid()) = 'admin'
    );

-- 2. Función de trigger para evitar que los no-admins actualicen columnas sensibles
CREATE OR REPLACE FUNCTION public.protect_user_sensitive_fields()
RETURNS TRIGGER AS $$
BEGIN
    -- Verificar si el usuario que realiza la acción es admin
    IF NOT EXISTS (
        SELECT 1 FROM public.usuarios
        WHERE id = auth.uid() AND role = 'admin'
    ) THEN
        -- Si no es admin, verificar si intenta cambiar campos restringidos
        IF (NEW.role IS DISTINCT FROM OLD.role) OR
           (NEW.is_store IS DISTINCT FROM OLD.is_store) OR
           (NEW.max_albums IS DISTINCT FROM OLD.max_albums) OR
           (NEW.max_pages IS DISTINCT FROM OLD.max_pages) OR
           (NEW.max_decks IS DISTINCT FROM OLD.max_decks) OR
           (NEW.max_cards_per_deck IS DISTINCT FROM OLD.max_cards_per_deck) OR
           (NEW.allowed_spirit_ids IS DISTINCT FROM OLD.allowed_spirit_ids) THEN
            RAISE EXCEPTION 'No tienes permisos para modificar campos administrativos (rol, límites, estado de tienda, etc).';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 3. Aplicar el trigger
DROP TRIGGER IF EXISTS on_user_update_protect_fields ON public.usuarios;
CREATE TRIGGER on_user_update_protect_fields
    BEFORE UPDATE ON public.usuarios
    FOR EACH ROW
    EXECUTE FUNCTION public.protect_user_sensitive_fields();
