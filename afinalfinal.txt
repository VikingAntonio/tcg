-- Consolidated SQL for Tournaments and Events System
-- Includes idempotent checks to prevent errors if relations already exist

-- 1. Tables for Tournaments
CREATE TABLE IF NOT EXISTS public.tournaments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES public.usuarios(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    tcg TEXT,
    tournament_type TEXT DEFAULT 'swiss',
    max_participants INTEGER DEFAULT 32,
    status TEXT DEFAULT 'planned', -- planned, active, finished
    registration_enabled BOOLEAN DEFAULT false,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    event_date TIMESTAMP WITH TIME ZONE,
    description TEXT,
    image_url TEXT
);

CREATE TABLE IF NOT EXISTS public.tournament_participants (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tournament_id UUID REFERENCES public.tournaments(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    deck_name TEXT,
    deck_list TEXT,
    player_id TEXT,
    points INTEGER DEFAULT 0,
    tiebreak_score DECIMAL DEFAULT 0.0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS public.tournament_matches (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tournament_id UUID REFERENCES public.tournaments(id) ON DELETE CASCADE,
    round INTEGER NOT NULL,
    player1_id UUID REFERENCES public.tournament_participants(id) ON DELETE CASCADE,
    player2_id UUID REFERENCES public.tournament_participants(id) ON DELETE CASCADE,
    result TEXT DEFAULT 'pending', -- p1_win, p2_win, draw, pending
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- 2. Tables for General Events
CREATE TABLE IF NOT EXISTS public.events (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES public.usuarios(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    description TEXT,
    event_date TIMESTAMP WITH TIME ZONE,
    image_url TEXT,
    is_public BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- 3. Update Usuarios table with plan-based limits
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS max_tournaments INTEGER DEFAULT 1;
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS has_tournaments BOOLEAN DEFAULT true;
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS max_events INTEGER DEFAULT 1;
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS has_events BOOLEAN DEFAULT true;

-- 4. Row Level Security (RLS) Configuration
ALTER TABLE public.tournaments ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournament_participants ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournament_matches ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.events ENABLE ROW LEVEL SECURITY;

-- Helper to safely create policies
DO $$
BEGIN
    -- Tournaments
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Users can manage their own tournaments') THEN
        CREATE POLICY "Users can manage their own tournaments" ON public.tournaments FOR ALL USING (auth.uid() = user_id);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Public can view tournaments') THEN
        CREATE POLICY "Public can view tournaments" ON public.tournaments FOR SELECT USING (true);
    END IF;

    -- Participants
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Users can manage participants of their tournaments') THEN
        CREATE POLICY "Users can manage participants of their tournaments" ON public.tournament_participants FOR ALL USING (
            EXISTS (SELECT 1 FROM public.tournaments WHERE tournaments.id = tournament_participants.tournament_id AND tournaments.user_id = auth.uid())
        );
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Public can view participants') THEN
        CREATE POLICY "Public can view participants" ON public.tournament_participants FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Public can register themselves') THEN
        CREATE POLICY "Public can register themselves" ON public.tournament_participants FOR INSERT WITH CHECK (true);
    END IF;

    -- Matches
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Users can manage matches of their tournaments') THEN
        CREATE POLICY "Users can manage matches of their tournaments" ON public.tournament_matches FOR ALL USING (
            EXISTS (SELECT 1 FROM public.tournaments WHERE tournaments.id = tournament_matches.tournament_id AND tournaments.user_id = auth.uid())
        );
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Public can view matches') THEN
        CREATE POLICY "Public can view matches" ON public.tournament_matches FOR SELECT USING (true);
    END IF;

    -- Events
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Users can manage their own events') THEN
        CREATE POLICY "Users can manage their own events" ON public.events FOR ALL USING (auth.uid() = user_id);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Public can view events') THEN
        CREATE POLICY "Public can view events" ON public.events FOR SELECT USING (true);
    END IF;
END
$$;
