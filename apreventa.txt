-- Migration to add preorders table
CREATE TABLE IF NOT EXISTS public.preorders (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID REFERENCES public.usuarios(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    image_url TEXT,
    price TEXT,
    payment_deadline TEXT, -- Fecha límite de pago
    tcg TEXT, -- 'yugioh', 'pokemon', 'onepiece', etc.
    is_public BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT now()
);

-- Enable RLS
ALTER TABLE public.preorders ENABLE ROW LEVEL SECURITY;

-- Policies
CREATE POLICY "Public preorders are viewable by everyone"
ON public.preorders FOR SELECT
USING (is_public = true);

CREATE POLICY "Users can manage their own preorders"
ON public.preorders FOR ALL
USING (auth.uid() = user_id);
