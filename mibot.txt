-- SQL para el sistema de mensajes del Chatbot/Acompañante
-- Ejecutar esto en el SQL Editor de Supabase

-- 1. Tabla de mensajes configurables
CREATE TABLE IF NOT EXISTS public.bot_messages (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    content TEXT NOT NULL,
    type TEXT NOT NULL, -- 'store_hours', 'store_location', 'album_link', 'pre_sales', 'promo', 'custom_1', 'custom_2'
    view_type TEXT DEFAULT 'public', -- 'public', 'admin', 'both'
    redirect_url TEXT, -- URL o ID de album para navegación
    display_duration INTEGER DEFAULT 5, -- segundos
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id, type) -- Permite tener un mensaje por cada tipo por usuario
);

-- RLS para bot_messages
ALTER TABLE public.bot_messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Los usuarios pueden ver mensajes de otros (publicos)"
ON public.bot_messages FOR SELECT
USING (true);

CREATE POLICY "Los usuarios pueden gestionar sus propios mensajes"
ON public.bot_messages FOR ALL
USING (auth.uid() = user_id);

-- 2. Columnas adicionales en la tabla usuarios para info de la tienda
ALTER TABLE public.usuarios
ADD COLUMN IF NOT EXISTS horario TEXT,
ADD COLUMN IF NOT EXISTS ubicacion TEXT,
ADD COLUMN IF NOT EXISTS website TEXT;

-- Nota: Estas columnas permiten que el bot extraiga información real del negocio.
