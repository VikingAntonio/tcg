-- SQL para el sistema de mensajes del Chatbot/Acompañante
-- Ejecutar esto en el SQL Editor de Supabase para corregir errores de columnas

-- 1. Eliminar la tabla si existe para asegurar una creación limpia con todas las columnas
DROP TABLE IF EXISTS public.bot_messages CASCADE;

-- 2. Crear la tabla de mensajes configurables
CREATE TABLE public.bot_messages (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID NOT NULL, -- Referencia al ID del usuario
    content TEXT NOT NULL,
    type TEXT NOT NULL, -- 'store_hours', 'store_location', 'album_link', 'pre_sales', 'promo', 'custom_1', 'custom_2'
    view_type TEXT DEFAULT 'public', -- 'public', 'admin', 'both'
    redirect_url TEXT, -- URL o ID de album para navegación
    duration INTEGER DEFAULT 5, -- Duración del globo en segundos (coincide con el frontend)
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id, type)
);

-- 3. Habilitar RLS
ALTER TABLE public.bot_messages ENABLE ROW LEVEL SECURITY;

-- 4. Políticas de Seguridad (Asegurarse de que coincidan con tu sistema)
CREATE POLICY "Mensajes visibles para todos"
ON public.bot_messages FOR SELECT
USING (true);

CREATE POLICY "Usuarios gestionan sus propios mensajes"
ON public.bot_messages FOR ALL
USING (auth.uid() = user_id);

-- 5. Columnas adicionales en la tabla usuarios (Si no existen)
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS horario TEXT;
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS ubicacion TEXT;
ALTER TABLE public.usuarios ADD COLUMN IF NOT EXISTS website TEXT;
