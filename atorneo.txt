-- SQL para la gestión de Torneos
-- Este archivo contiene la estructura para torneos, participantes y emparejamientos

-- Tabla de Torneos
CREATE TABLE IF NOT EXISTS public.tournaments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.usuarios(id) ON DELETE CASCADE NOT NULL,
    name TEXT NOT NULL,
    tcg TEXT NOT NULL, -- yugioh, pokemon, one_piece, lorcana, etc.
    description TEXT,
    tournament_type TEXT DEFAULT 'swiss', -- swiss, single_elimination, round_robin
    status TEXT DEFAULT 'planned', -- planned, active, finished
    registration_enabled BOOLEAN DEFAULT false,
    max_participants INTEGER,
    start_date TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabla de Participantes
CREATE TABLE IF NOT EXISTS public.tournament_participants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tournament_id UUID REFERENCES public.tournaments(id) ON DELETE CASCADE NOT NULL,
    user_id UUID REFERENCES public.usuarios(id) ON DELETE SET NULL,
    name TEXT NOT NULL,
    deck_name TEXT,
    deck_list TEXT, -- Almacena la lista de cartas
    player_id TEXT, -- ID oficial de TCG (Konami ID, etc)
    points INTEGER DEFAULT 0,
    tiebreak_score DECIMAL DEFAULT 0, -- OMWP o similar
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabla de Encuentros (Matches)
CREATE TABLE IF NOT EXISTS public.tournament_matches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tournament_id UUID REFERENCES public.tournaments(id) ON DELETE CASCADE NOT NULL,
    round INTEGER NOT NULL,
    player1_id UUID REFERENCES public.tournament_participants(id) ON DELETE CASCADE,
    player2_id UUID REFERENCES public.tournament_participants(id) ON DELETE CASCADE, -- NULL si es BYE
    result TEXT DEFAULT 'pending', -- p1_win, p2_win, draw, pending
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Extender tabla usuarios para límites de torneos
ALTER TABLE public.usuarios
ADD COLUMN IF NOT EXISTS max_tournaments INTEGER DEFAULT 1,
ADD COLUMN IF NOT EXISTS has_tournaments BOOLEAN DEFAULT TRUE;

-- Actualizar límites por rol

UPDATE public.usuarios SET max_tournaments = 10 WHERE role = 'premium';
UPDATE public.usuarios SET max_tournaments = 999 WHERE role IN ('admin', 'admin_store');

-- Habilitar RLS
ALTER TABLE public.tournaments ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournament_participants ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournament_matches ENABLE ROW LEVEL SECURITY;

-- Políticas para Tournaments
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Torneos son visibles por todos') THEN
        CREATE POLICY "Torneos son visibles por todos" ON public.tournaments FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar sus propios torneos') THEN
        CREATE POLICY "Usuarios pueden gestionar sus propios torneos" ON public.tournaments FOR ALL USING (auth.uid() = user_id);
    END IF;
END $$;

-- Políticas para Participantes
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Participantes son visibles por todos') THEN
        CREATE POLICY "Participantes son visibles por todos" ON public.tournament_participants FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar participantes de sus torneos') THEN
        CREATE POLICY "Usuarios pueden gestionar participantes de sus torneos" ON public.tournament_participants
            FOR ALL USING (
                EXISTS (
                    SELECT 1 FROM public.tournaments
                    WHERE tournaments.id = tournament_participants.tournament_id AND tournaments.user_id = auth.uid()
                )
            );
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Publico puede registrarse si habilitado') THEN
        CREATE POLICY "Publico puede registrarse si habilitado" ON public.tournament_participants
            FOR INSERT WITH CHECK (
                EXISTS (
                    SELECT 1 FROM public.tournaments
                    WHERE tournaments.id = tournament_id AND tournaments.registration_enabled = true
                )
            );
    END IF;
END $$;

-- Políticas para Matches
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Matches son visibles por todos') THEN
        CREATE POLICY "Matches son visibles por todos" ON public.tournament_matches FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar matches de sus torneos') THEN
        CREATE POLICY "Usuarios pueden gestionar matches de sus torneos" ON public.tournament_matches
            FOR ALL USING (
                EXISTS (
                    SELECT 1 FROM public.tournaments
                    WHERE tournaments.id = tournament_matches.tournament_id AND tournaments.user_id = auth.uid()
                )
            );
    END IF;
END $$;

-- Índices
CREATE INDEX IF NOT EXISTS idx_tournaments_user_id ON public.tournaments (user_id);
CREATE INDEX IF NOT EXISTS idx_participants_tournament_id ON public.tournament_participants (tournament_id);
CREATE INDEX IF NOT EXISTS idx_matches_tournament_id ON public.tournament_matches (tournament_id);
