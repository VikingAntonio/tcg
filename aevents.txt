-- SQL para la gestión de Eventos y Torneos
-- Este archivo contiene la estructura para eventos generales y funcionalidad de torneos integrada

-- Tabla de Eventos (incluye torneos)
CREATE TABLE IF NOT EXISTS public.events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.usuarios(id) ON DELETE CASCADE NOT NULL,
    name TEXT NOT NULL,
    tcg TEXT, -- yugioh, pokemon, one_piece, lorcana, etc. (Opcional para eventos generales)
    description TEXT,
    promo_details TEXT, -- Datos para promocionar el evento
    image_url TEXT, -- Imagen del evento
    event_date TIMESTAMP WITH TIME ZONE,
    status TEXT DEFAULT 'planned', -- planned, active, finished
    is_tournament BOOLEAN DEFAULT false,
    is_featured BOOLEAN DEFAULT false,
    tournament_type TEXT DEFAULT 'swiss', -- swiss, single_elimination, round_robin
    registration_enabled BOOLEAN DEFAULT false,
    max_participants INTEGER,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabla de Participantes (para eventos con torneo)
CREATE TABLE IF NOT EXISTS public.event_participants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID REFERENCES public.events(id) ON DELETE CASCADE NOT NULL,
    user_id UUID REFERENCES public.usuarios(id) ON DELETE SET NULL,
    name TEXT NOT NULL,
    deck_name TEXT,
    deck_list TEXT,
    player_id TEXT,
    points INTEGER DEFAULT 0,
    tiebreak_score DECIMAL DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabla de Encuentros (Matches)
CREATE TABLE IF NOT EXISTS public.event_matches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID REFERENCES public.events(id) ON DELETE CASCADE NOT NULL,
    round INTEGER NOT NULL,
    player1_id UUID REFERENCES public.event_participants(id) ON DELETE CASCADE,
    player2_id UUID REFERENCES public.event_participants(id) ON DELETE CASCADE,
    result TEXT DEFAULT 'pending', -- p1_win, p2_win, draw, pending
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Extender tabla usuarios para límites de eventos
ALTER TABLE public.usuarios
ADD COLUMN IF NOT EXISTS max_events INTEGER DEFAULT 1,
ADD COLUMN IF NOT EXISTS has_events BOOLEAN DEFAULT TRUE;

-- Actualizar límites por rol
UPDATE public.usuarios SET max_events = 1 WHERE role IN ('starter', 'user');
UPDATE public.usuarios SET max_events = 5 WHERE role = 'premium';
UPDATE public.usuarios SET max_events = 999 WHERE role IN ('admin', 'admin_store');

-- Habilitar RLS
ALTER TABLE public.events ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.event_participants ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.event_matches ENABLE ROW LEVEL SECURITY;

-- Políticas para Events
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Eventos son visibles por todos') THEN
        CREATE POLICY "Eventos son visibles por todos" ON public.events FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar sus propios eventos') THEN
        CREATE POLICY "Usuarios pueden gestionar sus propios eventos" ON public.events FOR ALL USING (auth.uid() = user_id);
    END IF;
END $$;

-- Políticas para Participantes
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Participantes de eventos son visibles por todos') THEN
        CREATE POLICY "Participantes de eventos son visibles por todos" ON public.event_participants FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar participantes de sus eventos') THEN
        CREATE POLICY "Usuarios pueden gestionar participantes de sus eventos" ON public.event_participants
            FOR ALL USING (
                EXISTS (
                    SELECT 1 FROM public.events
                    WHERE events.id = event_participants.event_id AND events.user_id = auth.uid()
                )
            );
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Publico puede registrarse en eventos si habilitado') THEN
        CREATE POLICY "Publico puede registrarse en eventos si habilitado" ON public.event_participants
            FOR INSERT WITH CHECK (
                EXISTS (
                    SELECT 1 FROM public.events
                    WHERE events.id = event_id AND events.registration_enabled = true
                )
            );
    END IF;
END $$;

-- Políticas para Matches
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Matches de eventos son visibles por todos') THEN
        CREATE POLICY "Matches de eventos son visibles por todos" ON public.event_matches FOR SELECT USING (true);
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Usuarios pueden gestionar matches de sus eventos') THEN
        CREATE POLICY "Usuarios pueden gestionar matches de sus eventos" ON public.event_matches
            FOR ALL USING (
                EXISTS (
                    SELECT 1 FROM public.events
                    WHERE events.id = event_matches.event_id AND events.user_id = auth.uid()
                )
            );
    END IF;
END $$;

-- Índices
CREATE INDEX IF NOT EXISTS idx_events_user_id ON public.events (user_id);
CREATE INDEX IF NOT EXISTS idx_event_participants_event_id ON public.event_participants (event_id);
CREATE INDEX IF NOT EXISTS idx_event_matches_event_id ON public.event_matches (event_id);
